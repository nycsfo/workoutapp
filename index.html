  <script>

    let calendar;

    let selectedEvent = null;


    document.addEventListener('DOMContentLoaded', function() {

      const containerEl = document.getElementById('external-events');

      

      new FullCalendar.Draggable(containerEl, {

        itemSelector: '.fc-event-template',

        eventData: function(eventEl) {

          return {

            title: eventEl.dataset.title,

            backgroundColor: eventEl.dataset.color,

            borderColor: eventEl.dataset.color,

            extendedProps: { details: eventEl.dataset.details }

          };

        }

      });


      const calendarEl = document.getElementById('calendar');

      

      calendar = new FullCalendar.Calendar(calendarEl, {

        initialView: 'dayGridMonth',

        headerToolbar: {

          left: 'prev,next',

          center: 'title',

          right: 'today'

        },

        editable: true,

        droppable: true,

        events: JSON.parse(localStorage.getItem('scheduledWorkouts') || '[]'),

        

        eventReceive: function() { saveCalendarState(); },

        eventChange: function() { saveCalendarState(); },

        eventClick: function(info) { openModal(info.event); }

      });


      calendar.render();

    });


    function saveCalendarState() {

      const events = calendar.getEvents().map(e => ({

        id: e.id,

        title: e.title,

        start: e.startStr,

        backgroundColor: e.backgroundColor,

        borderColor: e.borderColor,

        extendedProps: e.extendedProps

      }));

      localStorage.setItem('scheduledWorkouts', JSON.stringify(events));

    }


    function openModal(event) {

      selectedEvent = event;

      document.getElementById('modalTitle').innerText = event.title;

      document.getElementById('modalDetails').innerText = event.extendedProps.details || 'No exercise details available.';

      document.getElementById('detailsModal').style.display = 'flex';

    }


    function closeModal() {

      document.getElementById('detailsModal').style.display = 'none';

      selectedEvent = null;

    }


    document.getElementById('btnDeleteEvent').addEventListener('click', function() {

      if (selectedEvent) {

        selectedEvent.remove();

        saveCalendarState();

        closeModal();

      }

    });

  </script>
